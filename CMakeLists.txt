   AND CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")

  find_program(VYRUZ_READELF readelf)

  endif()

endif()

# MSVC limits individual string literals; keep shader bytes unchanged in chunks.
string(LENGTH "${SMAA_SOURCE}" _smaa_length)
set(SMAA_SOURCE_CHUNKS "")
set(_smaa_offset 0)
while(_smaa_offset LESS _smaa_length)
  math(EXPR _smaa_remaining "${_smaa_length} - ${_smaa_offset}")
  set(_smaa_count 8000)
  if(_smaa_remaining LESS _smaa_count)
    set(_smaa_count ${_smaa_remaining})
  endif()
  string(SUBSTRING "${SMAA_SOURCE}" ${_smaa_offset} ${_smaa_count} _smaa_chunk)
  string(APPEND SMAA_SOURCE_CHUNKS "R\"SMAASOURCE(${_smaa_chunk})SMAASOURCE\",\n")
  math(EXPR _smaa_offset "${_smaa_offset} + ${_smaa_count}")
endwhile()
